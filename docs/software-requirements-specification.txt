Creator Business Hub - Software Requirements Specification
Technical Implementation Guide for Cursor AI
________________


PROJECT OVERVIEW
Product Name: Creator Business Hub
Version: 1.0 MVP
Target Completion: 12 weeks
Architecture: Next.js 14 Modular Monolith → Microservices
Deployment: Vercel + Supabase
________________


TECHNICAL STACK
Frontend
* Framework: Next.js 14 (App Router, TypeScript)
* Styling: Tailwind CSS + shadcn/ui
* State: Zustand
* Charts: Recharts
* Forms: React Hook Form + Zod
* Auth: Auth0 SDK
Backend
* Runtime: Node.js 18+ (TypeScript)
* API: Next.js API Routes (RESTful)
* Validation: Zod schemas
* Architecture: Service Layer Pattern
Database
* Primary: PostgreSQL 15+ (Supabase)
* Cache: Redis (Upstash)
* ORM: Supabase JS Client
Infrastructure
* Hosting: Vercel (serverless)
* Database: Supabase (managed Postgres)
* Monitoring: Sentry + Vercel Analytics
* Logging: Axiom
________________


DATABASE SCHEMA
Core Tables
users
sql
CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  auth0_id VARCHAR(255) UNIQUE NOT NULL,
  email VARCHAR(255) UNIQUE NOT NULL,
  full_name VARCHAR(255),
  business_name VARCHAR(255),
  entity_type VARCHAR(50) CHECK (entity_type IN ('sole_proprietor', 'llc', 's_corp')),
  tax_id VARCHAR(20),
  primary_currency VARCHAR(3) DEFAULT 'USD',
  timezone VARCHAR(50) DEFAULT 'America/New_York',
  country VARCHAR(2) DEFAULT 'US',
  state VARCHAR(2),
  subscription_tier VARCHAR(50) DEFAULT 'starter',
  subscription_status VARCHAR(50) DEFAULT 'trialing',
  onboarding_completed BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);


CREATE INDEX idx_users_auth0_id ON users(auth0_id);
CREATE INDEX idx_users_email ON users(email);
transactions
sql
CREATE TABLE transactions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  date DATE NOT NULL,
  amount DECIMAL(12,2) NOT NULL CHECK (amount > 0),
  currency VARCHAR(3) DEFAULT 'USD',
  type VARCHAR(20) NOT NULL CHECK (type IN ('income', 'expense')),
  category VARCHAR(100) NOT NULL,
  subcategory VARCHAR(100),
  source VARCHAR(100),
  source_id VARCHAR(255),
  source_metadata JSONB,
  description TEXT,
  vendor_name VARCHAR(255),
  tax_deductible BOOLEAN,
  tax_category VARCHAR(100),
  receipt_url TEXT,
  invoice_id UUID REFERENCES invoices(id),
  brand_deal_id UUID REFERENCES brand_deals(id),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);


CREATE INDEX idx_transactions_user_id ON transactions(user_id);
CREATE INDEX idx_transactions_date ON transactions(date DESC);
CREATE INDEX idx_transactions_user_date ON transactions(user_id, date DESC);
CREATE INDEX idx_transactions_type ON transactions(type);
brand_deals
sql
CREATE TABLE brand_deals (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  brand_name VARCHAR(255) NOT NULL,
  contact_name VARCHAR(255),
  contact_email VARCHAR(255),
  contact_phone VARCHAR(50),
  status VARCHAR(50) NOT NULL DEFAULT 'lead' 
    CHECK (status IN ('lead', 'negotiating', 'contracted', 'delivered', 'paid', 'lost')),
  deal_value DECIMAL(12,2),
  currency VARCHAR(3) DEFAULT 'USD',
  payment_terms VARCHAR(100),
  contract_url TEXT,
  contract_signed_date DATE,
  deliverables JSONB DEFAULT '[]'::JSONB,
  deal_date DATE,
  due_date DATE,
  completed_date DATE,
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);


CREATE INDEX idx_deals_user_id ON brand_deals(user_id);
CREATE INDEX idx_deals_status ON brand_deals(status);
CREATE INDEX idx_deals_user_status ON brand_deals(user_id, status);
invoices
sql
CREATE TABLE invoices (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  brand_deal_id UUID REFERENCES brand_deals(id),
  invoice_number VARCHAR(50) UNIQUE NOT NULL,
  invoice_date DATE NOT NULL,
  due_date DATE NOT NULL,
  bill_to_name VARCHAR(255) NOT NULL,
  bill_to_email VARCHAR(255),
  bill_to_address TEXT,
  line_items JSONB NOT NULL DEFAULT '[]'::JSONB,
  subtotal DECIMAL(12,2) NOT NULL,
  tax_rate DECIMAL(5,2) DEFAULT 0,
  tax_amount DECIMAL(12,2) DEFAULT 0,
  total DECIMAL(12,2) NOT NULL,
  currency VARCHAR(3) DEFAULT 'USD',
  status VARCHAR(50) DEFAULT 'draft' 
    CHECK (status IN ('draft', 'sent', 'viewed', 'paid', 'overdue', 'cancelled')),
  payment_method VARCHAR(50),
  payment_date DATE,
  payment_transaction_id UUID REFERENCES transactions(id),
  stripe_payment_link TEXT,
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);


CREATE INDEX idx_invoices_user_id ON invoices(user_id);
CREATE INDEX idx_invoices_status ON invoices(status);
CREATE INDEX idx_invoices_number ON invoices(invoice_number);

MODULE 2: Transaction Management
Priority: P0 (Critical)
Dependencies: Module 1
Estimated Time: Week 2-3
Requirements
1. Manual Transaction Entry
   * Create income/expense
   * Upload receipts
   * Categorize transactions
   * Edit/delete transactions
2. Transaction Listing
   * Filter by type, category, date
   * Pagination
   * Sort by date, amount
   * Search by description
3. Stripe Integration
   * OAuth connection flow
   * Webhook handling
   * Auto-sync transactions
   * Auto-categorization
API Endpoints
GET /api/transactions
* Query params: type, category, start_date, end_date, limit, offset
* Return paginated list
* Include total count
POST /api/transactions
* Create new transaction
* Upload receipt to Supabase Storage
* Auto-categorize based on rules
* Return created transaction
GET /api/transactions/[id]
* Return single transaction
* Include related entities (invoice, deal)
PATCH /api/transactions/[id]
* Update transaction
* Validate ownership
* Invalidate cache
DELETE /api/transactions/[id]
* Soft delete (set deleted_at)
* Validate ownership
* Invalidate cache
POST /api/integrations/stripe/connect
* Initiate Stripe Connect OAuth
* Store connection details
* Return redirect URL
POST /api/integrations/stripe/webhook
* Verify webhook signature
* Handle charge.succeeded
* Create transaction record
* Return 200 OK

MODULE 3: Brand Deal Management
Priority: P0 (Critical)
Dependencies: Module 1, Module 2
Estimated Time: Week 4
Requirements
1. Deal Pipeline
   * Kanban board (5 stages)
   * Drag-and-drop status updates
   * Deal creation/editing
   * Deal notes
2. Deliverables Tracking
   * Add deliverables to deal
   * Mark deliverables complete
   * Due date reminders
3. Deal → Transaction Link
   * Auto-create transaction when status = "paid"
   * Link existing transactions to deals
API Endpoints
GET /api/deals
* Query param: status (optional)
* Return deals grouped by status
* Sort by created_at DESC
POST /api/deals
* Create new deal
* Default status: "lead"
* Return created deal
GET /api/deals/[id]
* Return single deal
* Include transactions
* Include invoices
PATCH /api/deals/[id]
* Update deal
* If status changed to "paid", create transaction
* Validate ownership
* Return updated deal
DELETE /api/deals/[id]
* Soft delete
* Validate ownership

export async function requireAuth(req: Request): Promise<string> {
  const userId = req.headers.get('x-user-id');
  if (!userId) {
    throw new UnauthorizedError('Authentication required');
  }
  return userId;
}


export function createAuthMiddleware() {
  return async (req: Request, next: () => Promise<Response>) => {
    try {
      const token = extractTokenFromHeader(req);
      const decoded = await verifyJWT(token);
      req.headers.set('x-user-id', decoded.user_id);
      return await next();
    } catch (err) {
      return new Response('Unauthorized', { status: 401 });
    }
  };
}

// lib/utils/cache.ts


export async function getCached<T>(
  key: string,
  fetcher: () => Promise<T>,
  ttl: number = 300
): Promise<T> {
  const cached = await redis.get(key);
  if (cached) {
    return JSON.parse(cached) as T;
  }
  
  const data = await fetcher();
  await redis.setex(key, ttl, JSON.stringify(data));
  return data;
}


export async function invalidateCache(pattern: string) {
  const keys = await redis.keys(pattern);
  if (keys.length > 0) {
    await redis.del(...keys);
  }
}


export function cacheKey(prefix: string, ...parts: string[]): string {
  return `${prefix}:${parts.join(':')}`;
}

export function handleError(error: unknown): Response {
  if (error instanceof AppError) {
    return new Response(
      JSON.stringify({
        error: error.message,
        statusCode: error.statusCode,
      }),
      {
        status: error.statusCode,
        headers: { 'Content-Type': 'application/json' },
      }
    );
  }
  
  if (error instanceof ZodError) {
    return new Response(
      JSON.stringify({
        error: 'Validation failed',
        details: error.errors,
      }),
      {
        status: 400,
        headers: { 'Content-Type': 'application/json' },
      }
    );
  }
  
  // Unknown error
  logger.error('Unhandled error', error as Error);
  
  return new Response(
    JSON.stringify({
      error: 'Internal server error',
    }),
    {
      status: 500,
      headers: { 'Content-Type': 'application/json' },
      }
    );
